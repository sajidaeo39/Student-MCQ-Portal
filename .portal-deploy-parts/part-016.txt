/* Reuse the original MCQ portal's shared Abacus counters and exact labels. */
const COUNTER_BASE="https://abacus.jasoncameron.dev";
const COUNTER_NAMESPACE="student-mcq-portal-sajidaeo39";
const COUNTER_KEYS={schools:"school_entries",students:"student_entries",tests:"tests_attempted",questions:"questions_attempted"};
const COUNTER_OFFSETS={schools:1,students:1,tests:7,questions:14};
let quizRecorded=false;
async function readCounter(key,id){const el=document.getElementById(id);try{const r=await fetch(`${COUNTER_BASE}/get/${COUNTER_NAMESPACE}/${COUNTER_KEYS[key]}?t=${Date.now()}`,{mode:"cors",cache:"no-store"});if(r.status===404){el.innerText=COUNTER_OFFSETS[key].toLocaleString();return true;}if(!r.ok)throw Error("Counter read failed");const d=await r.json();if(d.value===undefined)throw Error("Counter value missing");el.innerText=(COUNTER_OFFSETS[key]+Number(d.value)).toLocaleString();return true;}catch(e){el.innerText="—";return false;}}
async function loadPortalStats(){const el=document.getElementById("stats-status");el.innerText="Loading shared statistics...";const results=await Promise.all([readCounter("schools","total-schools"),readCounter("students","total-students"),readCounter("tests","tests-attempted"),readCounter("questions","questions-attempted")]);el.innerText=results.every(Boolean)?"Live shared statistics":"Statistics service unavailable. Please refresh or try another network.";}
async function hitCounter(key){const r=await fetch(`${COUNTER_BASE}/hit/${COUNTER_NAMESPACE}/${COUNTER_KEYS[key]}?t=${Date.now()}`,{mode:"cors",cache:"no-store"});if(!r.ok)throw Error("Counter update failed");const d=await r.json();if(d.value===undefined)throw Error("Counter value missing");return COUNTER_OFFSETS[key]+Number(d.value);}
async function hitCounterWithRetry(key){let err;for(let i=0;i<3;i++){try{return await hitCounter(key);}catch(e){err=e;await new Promise(r=>setTimeout(r,1100*(i+1)));}}throw err;}
async function saveAttemptStats(total,answered){if(quizRecorded)return;quizRecorded=true;const schoolEntered=document.getElementById("school-name").value.trim().length>0,studentEntered=document.getElementById("student-name").value.trim().length>0;try{const tests=await hitCounterWithRetry("tests");document.getElementById("tests-attempted").innerText=tests.toLocaleString();if(schoolEntered)document.getElementById("total-schools").innerText=(await hitCounterWithRetry("schools")).toLocaleString();if(studentEntered)document.getElementById("total-students").innerText=(await hitCounterWithRetry("students")).toLocaleString();const out=document.getElementById("questions-attempted");for(let i=0;i<answered;i++){out.innerText=(await hitCounterWithRetry("questions")).toLocaleString();if(i<answered-1)await new Promise(r=>setTimeout(r,350));}document.getElementById("stats-status").innerText="Statistics updated successfully";}catch(e){document.getElementById("stats-status").innerText="Some statistics could not sync. Please try again later.";}}
